```python
def entry_ring(e, by_id=None):
    """片区直接取 ring；功能区继承父片区；聚落用 type 本身（产业小镇／特色村落）。"""
    if e['type'] == '城市片区':
        return e['ring']
    if e['type'] == '功能区':
        return entry_ring(by_id[e['parent']], by_id)
    return e['type']
```

`monthly_band()` 换成两条线取大，并把 LVL1 专用钳改成全层通用钳（只顶起，不压低）：

```python
def monthly_band(anchor, level, shape, tier, basket):
    old_line = anchor / PARAMS['住所占比判据']          # 第1期原样保留
    eff_line = max(old_line, anchor + basket)          # 两条线取大
    base = old_line * PARAMS['层系数'][level] * shape * PARAMS['档位加成'][tier]
    lo = int(math.floor(base * PARAMS['带宽'][0] / PARAMS['月薪取整']) * PARAMS['月薪取整'])
    hi = int(math.ceil(base * PARAMS['带宽'][1] / PARAMS['月薪取整']) * PARAMS['月薪取整'])
    floor_line = int(math.ceil(eff_line / PARAMS['月薪取整']) * PARAMS['月薪取整'])
    return max(lo, floor_line), max(hi, floor_line)
```
