# 产业与就业层（第1期）实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 落地 20 大类／163 小类行业表、岗位五层骨架、公司五档参数与薪酬带算法，生成 64 个岗位锚点的公司骨架、岗位卡与薪酬带，并由十条门禁守住守恒与平衡。

**Architecture:** 单一可编辑源表 `数据/产业就业源表.json` ＋ `read_text` 读名录与定价文件 → `build_occupations.py` 确定性展开 → `产业就业正典.json`／`产业与就业总表.md`／`产业就业核验.json` → `verify_occupations.py` 十条判红 ＋ `negtest_occupations.py` 违例自检。生成器不内嵌任何正典常量。

**Tech Stack:** Python 3 标准库（json／pathlib／math／statistics／argparse／unittest／subprocess／tempfile）、Markdown 分卷与合订本、Node `export_docx.cjs`。

**Spec:** [钛湖城城市设定集/设计/2026-10-31-产业与就业层-design.md](../%E8%AE%BE%E8%AE%A1/2026-10-31-%E4%BA%A7%E4%B8%9A%E4%B8%8E%E5%B0%B1%E4%B8%9A%E5%B1%82-design.md)（六段设计已在对话中逐段批准，本文与 spec 同批读入）

## Global Constraints

- 工作树**不是 git 仓库**（`Is a git repository: false`），本计划所有任务以「校验点」步骤替代 commit：校验点＝跑该任务的验证命令并把结果落盘。

- 文件一律 UTF-8；中文文件名保持既有命名风格（`数据/xxx.json`、`工具/build_xxx.py`）。

- [x] **Step 5: 给 24 篇 S 原稿补岗位声明句**

`compile_book.py:31` 要求名录里的 `daily_workers` 在「位置与规模」或「人群与生活节奏」段有同值声明。三个文件各 10／10／4 处，把 `**人群与生活节奏**：常住` 替换为 `**人群与生活节奏**：另有<N>人为日常岗位人口，不并入常住口径；常住`，N 取上表值（S01＝7200 … S24＝460）。每篇一句、只改本句，不动其他字段。

**为什么不是「日常岗位规模为N人」**（实测过一次判红）：`FIELD_PATTERNS['population']` 第二条是裸的 `NUMBER\s*人(?!(?:为|是|次))`，「规模为7200人，」会被当成常住人数声明，24 条 S 全部报 `population: expected labeled value ..., found [7200.0, 24000.0]`。现措辞走 `daily_workers` 第二条模式 `N人(?:为|是)(?:园区)?日常岗位`，且 `人为` 被人口模式的负向查杀挡住，两栏各取一值。（功能区 S／F 不冲突：F 名录无 `population`，所以 F 原稿用「日常岗位规模为4000人，不代表常住居民」也能过。）

class SourceTable(unittest.TestCase):
    def setUp(self):
        self.src = json.loads(SOURCE.read_text(encoding='utf-8'))

    def test_class_and_subclass_counts_match_spec(self):
        self.assertEqual(len(self.src['大类']), 20)
        self.assertEqual(len(self.src['小类']), 163)
        counted = {}
        for s in self.src['小类']:
            counted[s['父大类']] = counted.get(s['父大类'], 0) + 1
        self.assertEqual(counted, SUBCLASS_COUNTS)

    def test_subclass_ids_are_sequential_within_class(self):
        seen = set()
        for s in self.src['小类']:
            cid, _, seq = s['id'].partition('-')
            self.assertEqual(s['父大类'], cid)
            self.assertLess(int(seq[1:]), SUBCLASS_COUNTS[cid] + 1)
            self.assertNotIn(s['id'], seen)
            seen.add(s['id'])

        for c in self.src['大类']:
            self.assertEqual(c['层制'], 4 if c['id'] in FOUR_LAYER_CLASSES else 5)

# -*- coding: utf-8 -*-
"""产业与就业层生成器：读名录＋参考房价租金＋产业就业源表，展开 64 区公司骨架、岗位卡与薪酬带。

正典只从三个 read_text 输入取得，脚本内不内嵌任何 ID、名称或坐标常量。
跑法：python 工具/build_occupations.py [--out-dir DIR]
"""
import json, math, statistics
from pathlib import Path

ROOT = Path(__file__).resolve().parents[1]
SOURCE = json.loads((ROOT / '数据' / '产业就业源表.json').read_text(encoding='utf-8'))
REGISTRY = json.loads((ROOT / '数据' / '城市空间名录.json').read_text(encoding='utf-8'))['entries']
PRICING = json.loads((ROOT / '数据' / '参考房价租金.json').read_text(encoding='utf-8'))['records']
PARAMS = SOURCE['参数']
BY_ID = {e['id']: e for e in REGISTRY}
PRICE_BY_ID = {r['id']: r for r in PRICING}


def household_area_proxy():
    values = [r['typical_dwelling']['avg_unit_gross_floor_area_m2'] for r in PRICING if 'typical_dwelling' in r]
    return statistics.median(values)


def district_anchors():
    """全部锚点：D 取其下属 R 的户均月租中位数；F 继承所属 D；S 用每平方米月租×户均面积代理。"""
    anchors = {}
    for d in [e for e in REGISTRY if e['type'] == '城市片区']:
        rents = [r['typical_dwelling']['monthly_rent_tt'] for r in PRICING
                 if r.get('parent') == d['id'] and 'typical_dwelling' in r]
        anchors[d['id']] = round(statistics.median(rents), 1)
    for f in [e for e in REGISTRY if e['type'] == '功能区']:
        anchors[f['id']] = anchors[f['parent']]
    proxy = household_area_proxy()
    for s in [e for e in REGISTRY if e['id'].startswith('S')]:
        anchors[s['id']] = round(PRICE_BY_ID[s['id']]['residential']['rent_tt_per_m2_month'] * proxy, 1)
    return anchors


def dwelling_anchors():
    """只保留可作为岗位锚点的 64 区（40F＋24S）；D 键只作继承，不对外暴露。"""
    return {k: v for k, v in district_anchors().items() if k[0] in 'FS'}


def monthly_band(anchor, level, shape, tier):
    red_line = anchor / PARAMS['住所占比判据']
    base = red_line * PARAMS['层系数'][level] * shape * PARAMS['档位加成'][tier]
    lo = int(math.floor(base * PARAMS['带宽'][0] / PARAMS['月薪取整']) * PARAMS['月薪取整'])
    hi = int(math.ceil(base * PARAMS['带宽'][1] / PARAMS['月薪取整']) * PARAMS['月薪取整'])
    if level == 'LVL1':
        # §5 第4条红线：一个全职岗位要撑得起住所锚 ÷ 0.40。形状系数 <1.00 的大类公式值在线下，带下限被钳平。
        lo = max(lo, int(math.ceil(red_line / PARAMS['月薪取整']) * PARAMS['月薪取整']))
    return lo, hi


def hourly(low, high):
    step = PARAMS['时薪取整']
    return (round(round(low / PARAMS['月标准工时'] / step) * step, 2),
            round(round(high / PARAMS['月标准工时'] / step) * step, 2))
```

def zone_workers():
    return {e['id']: int(e['daily_workers']) for e in REGISTRY if 'daily_workers' in e and e['id'][0] in 'FS'}


def per_company_cap(w):
    return max(1, min(PARAMS['单公司雇员上限'], math.floor(w * PARAMS['单公司占区配额上限'])))


BANDS = [(t['档'], t['雇员数'][0], t['雇员数'][1])
         for t in sorted(SOURCE['公司档'], key=lambda x: -x['雇员数'][0]) if t['档'] != 'T5']
NAMED_BANDS = [b for b in BANDS if b[0] != 'T1']
T5_BAND = tuple(next(t['雇员数'] for t in SOURCE['公司档'] if t['档'] == 'T5'))


def cut(budget, lo, hi):
    """把 budget 人切成尽量少的公司，每家落在 [lo, hi]；切不出返回 None，整笔余额交下一档回收。"""
    if budget < lo or hi < lo:
        return None
    n = max(1, math.ceil(budget / hi))
    if n > budget // lo:
        return None
    base, rem = divmod(budget, n)
    return [base + (1 if i < rem else 0) for i in range(n)]


def split_zone(w, cap, t4_left, t5_employees=0):
    """按档位人数配比切各区岗位池：T4→T2 依次认领，切不动的整笔向下结转，T1 收全部余数并合记一条。

    T1 在源表里 逐家具名=false，是统计口径而非一家个体户，所以这一行的人数可以越过个体户的 1—5 人档区间，
    单公司占区配额上限也不套在它身上——verify_occupations 的第 3 条门禁按 逐家具名 跳过统计条。
    """
    rows = [('T5', t5_employees)] if t5_employees else []
    pool = w - t5_employees
    carry, allocated = 0, 0
    for tier, lo, hi in NAMED_BANDS:
        hi = min(hi, cap)
        budget = carry + math.floor(pool * PARAMS['档位人数配比'][tier] + 0.5)
        if tier == 'T4':
            if t4_left <= 0:
                carry = budget
                continue
            budget = min(budget, t4_left * hi)
        chunks = cut(budget, lo, hi)
        if not chunks:
            carry = budget
            continue
        rows += [(tier, e) for e in chunks]
        carry, allocated = 0, allocated + sum(chunks)
    return rows + [('T1', pool - allocated)]


def t5_hosts():
    """每个大类恰好一家 T5，且一锚只放一家：按候选最少的类先占位，避免多类撞同一锚。"""
    zw = zone_workers()
    min_zone = math.ceil(T5_BAND[0] / PARAMS['单公司占区配额上限'])
    pools = {c['id']: sorted((z for z in c['可落锚点'] if z in zw and zw[z] >= min_zone),
                             key=lambda z: (-zw[z], z)) for c in SOURCE['大类']}
    for cid, cand in pools.items():
        if not cand:
            raise SystemExit(f'T5 无可用锚点：{cid}（需要 daily_workers ≥ {min_zone} 的可落锚点）')
    hosts, taken = {}, set()
    for cid in sorted(pools, key=lambda k: (len(pools[k]), k)):
        for z in pools[cid]:
            if z not in taken:
                hosts[cid] = z
                taken.add(z)
                break
        else:
            raise SystemExit(f'T5 锚点冲突：{cid} 的候选被占满，请回源表补可落锚点，不要放宽门禁')
    return hosts

def level_slots(emp, scheme, allowed):
    """最大余额法按层名额分布切人：整数名额之和恒等于雇员数，不产生 0 人层；最低层保底 1 人。"""
    quota = PARAMS['层名额分布'][f'{scheme}层']
    levels = [l for l in allowed if quota[l] > 0]
    raw = [(l, emp * quota[l]) for l in levels]
    counts = {l: math.floor(v) for l, v in raw}
    for l, _ in sorted(raw, key=lambda x: counts[x[0]] - x[1])[:emp - sum(counts.values())]:
        counts[l] += 1
    if counts[levels[0]] == 0:
        donor = max(levels, key=lambda l: counts[l])
        counts[donor] -= 1
        counts[levels[0]] = 1
    return [(l, counts[l]) for l in levels if counts[l] > 0]


NO_FLY = set(SOURCE['禁则']['无人机禁区'])
LOW_ALT = set(SOURCE['禁则']['低空作业小类'])


def subclass_for(cid, seq, zid):
    """轮转取小类；落点在净空禁区内时从轮转池里剔除低空作业小类，避免生成即判红。"""
    pool = SUBS[cid]
    if zid in NO_FLY:
        safe = [s for s in pool if s['id'] not in LOW_ALT]
        if safe:
            pool = safe
    return pool[seq % len(pool)]


def expand_companies():
    zw, anchors, hosts = zone_workers(), dwelling_anchors(), t5_hosts()
    host_by_anchor = {v: k for k, v in hosts.items()}
    companies = []
    for zid in sorted(zw, key=lambda x: (x[0], x[1:])):
        classes = [c for c in ZONE_TO_CLASSES.get(zid, []) if CLASS[c]['可招聘'] is True] or \
                  [c for c in ZONE_TO_CLASSES.get(zid, [])]
        w, cap = zw[zid], per_company_cap(zw[zid])
        t5_emp = min(T5_BAND[1], math.floor(w * PARAMS['单公司占区配额上限'])) if zid in host_by_anchor else 0
        t4_left = 0 if BY_ID[zid].get('type') == '特色村落' else PARAMS['T4每区家数上限']

            companies.append(make_company(zid, tier, emp, cid, k, anchors[zid]))
    return companies


def make_company(zid, tier, emp, cid, seq, anchor):
    layer_scheme = CLASS[cid]['层制']
    slots = level_slots(emp, layer_scheme, TIERS[tier]['可设岗位层'])

                      '用工形态': ['全职', '兼职', '轮班', '临时', '自由接单'],

"""产业与就业层闸门：十条判红全部落在数据层，任一失败即非零退出。跑法：python 工具/verify_occupations.py"""
import json, math
from pathlib import Path

ROOT = Path(__file__).resolve().parents[1]
SOURCE = json.loads((ROOT / '数据' / '产业就业源表.json').read_text(encoding='utf-8'))
REGISTRY = json.loads((ROOT / '数据' / '城市空间名录.json').read_text(encoding='utf-8'))['entries']


def issues(canon, source, registry):
    out, p = [], canon['参数']
    by_id = {e['id']: e for e in registry}
    bands = {t['档']: tuple(t['雇员数']) for t in source['公司档']}
    zw = {e['id']: int(e['daily_workers']) for e in registry if 'daily_workers' in e and e['id'][0] in 'FS'}
    by_zone = {}
    for c in canon['公司']:
        by_zone.setdefault(c['锚点'], []).append(c)
    subs = {s['id']: s for s in canon['行业']['小类']}

    # 1 逐区守恒：公司雇员之和必须等于名录日常岗位，锚点集合不得多也不得少
    for z, want in sorted(zw.items()):
        got = sum(c['雇员数'] for c in by_zone.get(z, []))
        if got != want:
            out.append(f'守恒: {z} 公司雇员 {got} != 名录日常岗位 {want}')
    if set(by_zone) != set(zw):
        out.append(f'守恒: 锚点集合不一致 {sorted(set(by_zone) ^ set(zw))[:6]}')

    # 2 T5 恰好每大类一家
    t5 = [c for c in canon['公司'] if c['档位'] == 'T5']
    if len(t5) != len(source['大类']):
        out.append(f'T5: 数量 {len(t5)} != 大类数 {len(source["大类"])}')
    if len({c['大类'] for c in t5}) != len(t5):
        out.append('T5: 同大类出现多家')

    # 3 档位区间、单公司占比、T4 家数与 T1 统计口径
    #   T1 在源表里 逐家具名=false，是每区一条的个体户统计口径，不套个体户 1—5 人档区间与占比上限
    ratio, t4_max = p['单公司占区配额上限'], p['T4每区家数上限']
    for c in canon['公司']:
        if not c['逐家具名']:
            continue
        lo, hi = bands[c['档位']]
        if not lo <= c['雇员数'] <= hi:
            out.append(f'档位区间: {c["id"]} {c["档位"]} 雇员 {c["雇员数"]} 不在 [{lo}, {hi}]')
        if c['雇员数'] > max(1, math.floor(zw[c['锚点']] * ratio)):
            out.append(f'占比: {c["id"]} 雇员 {c["雇员数"]} 超过所在区配额上限')
    for z, rows in sorted(by_zone.items()):
        n_t4 = sum(1 for x in rows if x['档位'] == 'T4')
        if n_t4 > t4_max:
            out.append(f'T4: {z} 有 {n_t4} 家，超过上限 {t4_max}')
        if n_t4 and by_id[z].get('type') == '特色村落':
            out.append(f'落点: {z} 特色村落没有专属功能区，不该有 T4')
        if sum(1 for x in rows if x['逐家具名'] is False) != 1:
            out.append(f'T1统计口径: {z} 应有且只应有 1 条')

    # 4 住所覆盖红线：LVL1 带下限不得低于住所锚 ÷ 判据
    for c in canon['公司']:
        for card in c['岗位卡']:
            if card['岗位层'] != 'LVL1':
                continue
            floor = canon['住所锚'][c['锚点']] / p['住所占比判据']
            if card['月薪带_钛币'][0] < floor:
                out.append(f'住所覆盖红线: {card["id"]} 月薪下限 {card["月薪带_钛币"][0]} < {floor:.0f}')

    # 5 层系数严格单调且不倒挂
    coef, order = p['层系数'], ['LVL1', 'LVL2', 'LVL3', 'LVL4', 'LVL5']
    for a, b in zip(order, order[1:]):
        if coef[b] <= coef[a]:
            out.append(f'层系数非单调: {a}->{b}')
    if coef['LVL5'] / coef['LVL1'] > 5.0:
        out.append('层系数: LVL5/LVL1 超过 5.0')

    # 6 起始年窗口，且卡片起始年必须等于所属小类
    win0, win1 = source['禁则']['起始年窗口']
    for s in canon['行业']['小类']:
        if not win0 <= s['起始城市年'] <= win1:
            out.append(f'起始年越窗: {s["id"]} {s["起始城市年"]}')
    for c in canon['公司']:
        for card in c['岗位卡']:
            sub = subs.get(card['行业小类'])
            if sub and card['起始城市年'] != sub['起始城市年']:
                out.append(f'起始年漂移: {card["id"]}')

    # 7 A08 职责禁则：关键词不得出现在层职责或生成岗位卡的小类名称里，背景小类不得生成卡
    for cid, words in source['禁则']['职责关键词'].items():
        texts = [l['职责'] for l in source['岗位层']]
        texts += [subs[k]['名称'] for k in subs if subs[k]['父大类'] == cid and subs[k]['生成岗位卡']]
        texts += [c['名称'] for c in canon['公司'] if c['大类'] == cid and c['名称']]
        for text in texts:
            for w in words:
                if w in text:
                    out.append(f'A08禁则: 职责或小类名称出现"{w}"（{text[:24]}）')
    for card in [x for c in canon['公司'] if c['大类'] == 'A08' for x in c['岗位卡']]:
        if card['行业小类'] in source['禁则']['A08背景小类']:
            out.append(f'A08禁则: 背景小类生成岗位卡 {card["id"]}')

    # 8 不可申请边界：公职大类整类 可招聘=false，两区不得留任何在池名额
    for c in canon['公司']:
        if c['大类'] in source['禁则']['不可申请大类'] and c['可招聘'] is not False:
            out.append(f'公职边界: {c["id"]} 可招聘未置 false')
    pool = sum(c['雇员数'] for c in canon['公司']
               if c['可招聘'] and c['锚点'] in source['禁则']['不可申请锚点'])
    if pool:
        out.append(f'公职边界: F001／F002 名额池仍有 {pool} 人在池内')

    # 9 字段黑名单：正典全文不得出现现实金融与前世设定字段
    blob = json.dumps(canon, ensure_ascii=False)
    for w in source['禁则']['字段黑名单']:
        if w in blob:
            out.append(f'字段黑名单: 出现"{w}"')

    # 10 无人机禁区：低空作业小类与含无人机字样的具名公司都不得落在净空受限聚落
    low_alt = set(source['禁则']['低空作业小类'])
    for zid in source['禁则']['无人机禁区']:
        for c in canon['公司']:
            if c['锚点'] != zid:
                continue
            if c['行业小类'] in low_alt:
                out.append(f'无人机禁空: {c["id"]} 低空作业小类 {c["行业小类"]} 落在 {zid}')
            if c['名称'] and any(k in c['名称'] for k in source['禁则']['无人机关键词']):
                out.append(f'无人机禁空: {c["id"]} 具名含无人机字样，落在 {zid}')
    return out


CHECKS = ['分区守恒', 'T5每大类一家', '档位区间与占比', '住所覆盖红线', '层系数单调',
          '起始年窗口', 'A08职责禁则', '公职不可申请边界', '字段黑名单', '无人机禁区']


def main():
    canon = json.loads((ROOT / '数据' / '产业就业正典.json').read_text(encoding='utf-8'))
    bad = issues(canon, SOURCE, REGISTRY)
    report = {'检查项': CHECKS, '检查项数': len(CHECKS), '失败数': len(bad), '失败明细': bad[:40],
              '判据': '十条判红；任一失败即非零退出，不允许空转判绿'}
    (ROOT / '数据' / '产业就业核验.json').write_text(json.dumps(report, ensure_ascii=False, indent=2),
                                                    encoding='utf-8')
    if bad:
        for b in bad[:20]:
            print('FAIL', b)
        raise SystemExit(1)
    print(f'全绿：十条判红，0 失败；公司 {canon["汇总"]["公司数"]} 家（具名 {canon["汇总"]["具名公司数"]}）'
          f'，岗位卡 {canon["汇总"]["岗位卡数"]} 张，雇员 {canon["汇总"]["雇员总数"]} 人')


if __name__ == '__main__':
    main()

# -*- coding: utf-8 -*-
"""产业与就业层闸门的违例自检：把闸门与三份数据整批复制到临时目录，逐条改坏正典，断言闸门必须判红并给出该条自己的措辞。

闸门只会打印"全绿"，无法自证它抓得住违例；本脚本按十条判据反向注入，只操作临时副本，
绝不改动被守护的仓库文件。跑法：python 工具/negtest_occupations.py
"""
import json
import shutil
import subprocess
import sys
import tempfile
from pathlib import Path

ROOT = Path(__file__).resolve().parents[1]
GATE = ROOT / '工具' / 'verify_occupations.py'
DATA = ['产业就业正典.json', '产业就业源表.json', '城市空间名录.json']
TMP_PREFIX = 'occneg-'


def canon_of(tmp):
    return json.loads((tmp / '数据' / '产业就业正典.json').read_text(encoding='utf-8'))


def first(canon, pred):
    return next(x for x in canon['公司'] if pred(x))


def cases():
    """返回 (用例名, 改动函数, 期望出现的判红字样)；改动只动正典文件，源表与名录保持原样。"""

    def conservation(c):
        c['公司'][0]['雇员数'] += 1

    def missing_anchor(c):
        anchor = c['公司'][0]['锚点']
        c['公司'] = [x for x in c['公司'] if x['锚点'] != anchor]

    def no_t5(c):
        drop = first(c, lambda x: x['档位'] == 'T5')
        c['公司'].remove(drop)

    def duplicate_t5(c):
        t5 = first(c, lambda x: x['档位'] == 'T5')
        c['公司'].append(dict(t5, id='CORP-X-99'))

    def out_of_band(c):
        row = first(c, lambda x: x['档位'] == 'T3')
        row['雇员数'] = row['雇员数'] + 400

    def over_quota_and_village(c):
        # 挑岗位规模最小的特色村落：它的单公司配额上限必然装不下 T4 的 250 人门槛，占比与落点同时成立。
        reg = json.loads((ROOT / '数据' / '城市空间名录.json').read_text(encoding='utf-8'))['entries']
        village = min((e['id'] for e in reg if e.get('type') == '特色村落' and 'daily_workers' in e),
                      key=lambda z: next(e['daily_workers'] for e in reg if e['id'] == z))
        row = first(c, lambda x: x['锚点'] == village and x['逐家具名'])
        row['档位'], row['雇员数'], row['逐家具名'] = 'T4', 250, True

    def below_red_line(c):
        card = next(x for comp in c['公司'] for x in comp['岗位卡'] if x['岗位层'] == 'LVL1')
        card['月薪带_钛币'][0] = 1

    def flat_coefficient(c):
        c['参数']['层系数']['LVL3'] = c['参数']['层系数']['LVL2']

    def year_out_of_window(c):
        c['行业']['小类'][0]['起始城市年'] = 2080

    def year_drift(c):
        card = first(c, lambda x: x['岗位卡'])['岗位卡'][0]
        card['起始城市年'] = card['起始城市年'] + 3

    def a08_wording(c):
        sub = next(s for s in c['行业']['小类']
                   if s['父大类'] == 'A08' and s['生成岗位卡'])
        sub['名称'] = sub['名称'] + '授信'

    def a08_background_card(c):
        comp = first(c, lambda x: x['大类'] == 'A08')
        comp['岗位卡'][0]['行业小类'] = 'A08-B03'

    def public_posted(c):
        first(c, lambda x: x['大类'] == 'A01')['可招聘'] = True

    def blacklisted_field(c):
        c['公司'][0]['估值'] = '12亿'

    def drone_in_no_fly(c):
        comp = first(c, lambda x: x['锚点'] == 'S02')
        comp['行业小类'], comp['名称'] = 'A15-B03', '云禾植保无人机服务队'

    def t1_statistic_removed(c):
        row = first(c, lambda x: not x['逐家具名'])
        row['逐家具名'] = True

    def five_t4_in_one_zone(c):
        # 落在 T4 已达上限的那一区，否则加一家也不越界，闸门抓不到这条判据。
        counts = {}
        for x in c['公司']:
            if x['档位'] == 'T4':
                counts[x['锚点']] = counts.get(x['锚点'], 0) + 1
        anchor = max(counts, key=counts.get)
        donor = first(c, lambda x: x['锚点'] == anchor and x['档位'] == 'T4')
        c['公司'].append(dict(donor, id='CORP-X-98'))

    def coefficient_gap(c):
        c['参数']['层系数']['LVL5'] = 5.5

    return [
        ('逐区守恒人数被改', conservation, '守恒'),
        ('整区公司缺失', missing_anchor, '守恒'),
        ('T5 少一家', no_t5, 'T5'),
        ('T5 同大类两家', duplicate_t5, 'T5'),
        ('档位越界', out_of_band, '档位区间'),
        ('超所在区配额且落特色村落', over_quota_and_village, '占比'),
        ('T1 统计条被改成具名', t1_statistic_removed, 'T1统计口径'),
        ('每区第五家 T4', five_t4_in_one_zone, 'T4:'),
        ('LVL1 跌破住所红线', below_red_line, '住所覆盖红线'),
        ('层系数非单调', flat_coefficient, '层系数非单调'),
        ('层系数顶端超 5 倍', coefficient_gap, '层系数: LVL5/LVL1'),
        ('起始年越窗', year_out_of_window, '起始年越窗'),
        ('卡片起始年漂移', year_drift, '起始年漂移'),
        ('A08 职责关键词', a08_wording, 'A08禁则'),
        ('A08 背景小类生成卡', a08_background_card, 'A08禁则'),
        ('公职岗位进招募池', public_posted, '公职边界'),
        ('字段黑名单', blacklisted_field, '字段黑名单'),
        ('无人机落进净空禁区', drone_in_no_fly, '无人机禁空'),
    ]


def run_gate(tmp):
    r = subprocess.run([sys.executable, str(tmp / '工具' / 'verify_occupations.py')],
                       cwd=tmp, capture_output=True, text=True)
    return r.returncode, r.stdout + r.stderr


def main():
    results, failures = [], []
    for name, mutate, expected in cases():
        tmp = Path(tempfile.mkdtemp(prefix=TMP_PREFIX))
        try:
            (tmp / '工具').mkdir()
            (tmp / '数据').mkdir()
            shutil.copy2(GATE, tmp / '工具' / 'verify_occupations.py')
            for f in DATA:
                shutil.copy2(ROOT / '数据' / f, tmp / '数据' / f)
            rc0, _ = run_gate(tmp)
            canon = canon_of(tmp)
            mutate(canon)
            (tmp / '数据' / '产业就业正典.json').write_text(
                json.dumps(canon, ensure_ascii=False, indent=2), encoding='utf-8')
            rc, out = run_gate(tmp)
            red = rc != 0 and expected in out
            ok = rc0 == 0
            results.append(dict(case=name, rc=rc, red=red, clean_before=ok))
            print(('OK ' if red and ok else '!! ') + f'{name}：夹具原样 rc={rc0}，注入后 rc={rc}，期望字样「{expected}」')
            if not red or not ok:
                failures.append(name)
        finally:
            shutil.rmtree(tmp, ignore_errors=True)
    leftover = [p for p in Path(tempfile.gettempdir()).glob(TMP_PREFIX + '*') if p.is_dir()]
    print(json.dumps(dict(cases=len(results), red=len(results) - len(failures),
                          not_red=len(failures), fixture_leftover=len(leftover)), ensure_ascii=False))
    if failures or leftover:
        sys.exit(1)
    print(f'产业就业闸门违例自检通过：{len(results)} 个违例全部判红且措辞对得上，夹具零残留')


if __name__ == '__main__':
    main()
```
